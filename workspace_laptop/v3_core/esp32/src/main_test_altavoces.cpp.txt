/**
 * 🔊 PRUEBA DE ALTAVOCES ESP32 (MAX98357A)
 * 
 * Este archivo prueba SOLAMENTE los altavoces.
 * Recibe audio estéreo crudo (16kHz, 16-bit) por WebSocket
 * y lo inyecta directamente al amplificador I2S.
 */

#include <Arduino.h>
#include <WiFi.h>
#include <ArduinoOTA.h>
#include <ArduinoWebsockets.h>
#include <driver/i2s.h>

const char* WIFI_SSID = "robot_mesero_2.4G";
const char* WIFI_PASS = "";
const char* WS_HOST   = "192.168.1.2";
const int   WS_PORT   = 3006;
const char* WS_PATH   = "/ws/speaker";

// ── PINOUT I2S (ALTAVOCES) ───────────────────────────────────
#define I2S_SPK_BCLK    26
#define I2S_SPK_LRC     25
#define I2S_SPK_DOUT    22

#define SAMPLE_RATE     16000
#define BUFFER_SAMPLES  512

using namespace websockets;
WebsocketsClient wsClient;

volatile bool wsConnected = false;
unsigned long lastReconnect = 0;

void initI2SSpeaker() {
    i2s_config_t i2sSpkConfig = {
        .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_TX),
        .sample_rate = SAMPLE_RATE,
        .bits_per_sample = I2S_BITS_PER_SAMPLE_16BIT,
        .channel_format = I2S_CHANNEL_FMT_RIGHT_LEFT,
        .communication_format = I2S_COMM_FORMAT_STAND_I2S,
        .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
        .dma_buf_count = 8,
        .dma_buf_len = BUFFER_SAMPLES,
        .use_apll = false,
        .tx_desc_auto_clear = true,
        .fixed_mclk = 0
    };
    i2s_pin_config_t i2sSpkPins = {
        .bck_io_num = I2S_SPK_BCLK,
        .ws_io_num = I2S_SPK_LRC,
        .data_out_num = I2S_SPK_DOUT,
        .data_in_num = I2S_PIN_NO_CHANGE
    };
    i2s_driver_install(I2S_NUM_1, &i2sSpkConfig, 0, NULL);
    i2s_set_pin(I2S_NUM_1, &i2sSpkPins);
    Serial.println("[I2S] Altavoces inicializados (16kHz estéreo)");
}

void connectWiFi() {
    WiFi.mode(WIFI_STA);
    WiFi.begin(WIFI_SSID, WIFI_PASS);
    Serial.printf("[WiFi] Conectando");
    while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
    Serial.printf("\n[WiFi] Conectado. IP: %s\n", WiFi.localIP().toString().c_str());
}

void connectWS() {
    if (wsConnected || millis() - lastReconnect < 3000) return;
    lastReconnect = millis();

    String url = String("ws://") + WS_HOST + ":" + WS_PORT + WS_PATH;
    Serial.printf("[WS] Conectando a %s\n", url.c_str());

    wsClient = WebsocketsClient();
    wsClient.onMessage([](WebsocketsMessage msg) {
        if (msg.isBinary()) {
            size_t written;
            i2s_write(I2S_NUM_1, msg.c_str(), msg.length(), &written, portMAX_DELAY);
        }
    });
    wsClient.onEvent([](WebsocketsEvent event, String data) {
        if (event == WebsocketsEvent::ConnectionOpened) {
            wsConnected = true;
            Serial.println("[WS] ¡Conectado al servidor de audio!");
        } else if (event == WebsocketsEvent::ConnectionClosed) {
            wsConnected = false;
            Serial.println("[WS] Desconectado");
        }
    });

    wsClient.connect(url);
}

void setup() {
    Serial.begin(115200);
    delay(500);
    Serial.println("\n═══════════════════════════════════════");
    Serial.println("  PRUEBA DE ALTAVOCES ESP32");
    Serial.println("═══════════════════════════════════════");

    connectWiFi();

    ArduinoOTA.setHostname("chipi-esp32-test");
    ArduinoOTA.setPassword("");
    ArduinoOTA.begin();

    initI2SSpeaker();
}

void loop() {
    ArduinoOTA.handle();
    if (WiFi.status() == WL_CONNECTED) {
        if (!wsConnected) connectWS();
        wsClient.poll();
    }
}
